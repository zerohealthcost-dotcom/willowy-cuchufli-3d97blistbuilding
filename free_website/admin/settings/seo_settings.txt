<script>
  const consentRequiredCountries = [
    "AT", "BE", "BG", "HR", "CY", "CZ", "DK", "EE", "FI", "FR", "DE",
    "GR", "HU", "IS", "IE", "IT", "LV", "LI", "LT", "LU", "MT", "NL",
    "NO", "PL", "PT", "RO", "SK", "SI", "ES", "SE", "CH", "GB",
    "CA", "US"
  ];

  function getCountryAndCheckConsent() {
    if (localStorage.getItem("cookieConsent")) return;

    fetch('https://ipapi.co/json/')
      .then(response => response.json())
      .then(data => {
        const countryCode = data.country;
        if (consentRequiredCountries.includes(countryCode)) {
          document.getElementById('cookie-consent-banner').style.display = 'block';
        }
      })
      .catch(error => {
        console.warn("Geo lookup failed:", error);
      });
  }

  function acceptCookies() {
    localStorage.setItem("cookieConsent", "true");
    document.getElementById('cookie-consent-banner').style.display = 'none';
  }

  window.onload = getCountryAndCheckConsent;
</script>
